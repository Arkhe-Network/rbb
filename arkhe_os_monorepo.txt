use arkhe_agents::agent::AgentManager;

use arkhe_llm::inference::InferenceEngine;

use tokio::sync::Mutex;

/// AgiCoordinator
pub struct AgiCoordinator {
    /// Agents
    pub agents: Arc<Mutex<AgentManager>>,
    /// Scheduler
    pub scheduler: Arc<Mutex<IntentScheduler>>,
    /// LLM
    pub llm: Arc<dyn InferenceEngine>,
    /// Signer
    pub signer: MldsaSigner,
}

impl AgiCoordinator {
    /// New
    pub fn new(llm: Arc<dyn InferenceEngine>) -> Self {
        Self {
            agents: Arc::new(Mutex::new(AgentManager::new())),
            scheduler: Arc::new(Mutex::new(IntentScheduler::new())),
            llm,
            signer: MldsaSigner::generate(),
        }
    }

    /// Run
    pub async fn run(&self) {
        if let Some(intent) = self.scheduler.lock().await.schedule_next() {
            println!("Executing intent: {:?}", intent);
        }
    }
}

use crate::capsule::Capsule;
use arkhe_core::types::{AgentId, CapabilityToken, Intent};
use std::collections::HashMap;

/// Agent
pub struct Agent {
    /// ID
    pub id: AgentId,
    /// Name
    pub name: String,
    /// Capsule
    pub capsule: Capsule,
    /// Capabilities
    pub capabilities: Vec<CapabilityToken>,
    /// Intent queue
    pub intent_queue: Vec<Intent>,
}

impl Agent {
    /// New
    pub fn new(id: AgentId, name: &str) -> Self {
        Self {
            id,
            name: name.to_string(),
            capsule: Capsule::new(id),
            capabilities: Vec::new(),
            intent_queue: Vec::new(),
        }
    }

    /// Submit intent
    pub fn submit_intent(&mut self, intent: Intent) {
        self.intent_queue.push(intent);
    }
}

/// AgentManager
pub struct AgentManager {
    agents: HashMap<AgentId, Agent>,
    next_id: u64,
}

impl Default for AgentManager {
    fn default() -> Self {
        Self::new()
    }
}

impl AgentManager {
    /// New
    pub fn new() -> Self {
        Self {
            agents: HashMap::new(),
            next_id: 0,
        }
    }

    /// Create agent
    pub fn create_agent(&mut self, name: &str) -> AgentId {
        let id = AgentId(self.next_id);
        self.next_id += 1;
        self.agents.insert(id, Agent::new(id, name));
        id
    }

    /// Get agent
    pub fn get_agent(&self, id: AgentId) -> Option<&Agent> {
        self.agents.get(&id)
    }
}

/// IntentScheduler
#[derive(Default)]
pub struct IntentScheduler {
    queue: VecDeque<Intent>,
}

impl IntentScheduler {
    /// New
    pub fn new() -> Self {
        Self {
            queue: VecDeque::new(),
        }
    }

    /// Submit
    pub fn submit(&mut self, intent: Intent) {
        self.queue.push_back(intent);
    }

    /// Schedule next
    pub fn schedule_next(&mut self) -> Option<Intent> {
        self.queue.pop_front()
    }
}

use ml_dsa::{MlDsa87, signature::{Signer, Verifier, SignatureEncoding, Keypair}};
use ml_dsa::{SigningKey, VerifyingKey, Signature, KeyExport, Generate};
use arkhe_core::error::{KernelError, Result};

/// MldsaSigner
pub struct MldsaSigner {
    signing_key: SigningKey<MlDsa87>,
    verifying_key: VerifyingKey<MlDsa87>,
}

impl MldsaSigner {
    /// Generate
    pub fn generate() -> Self {
        let sk = ml_dsa::SigningKey::<MlDsa87>::generate();
        let vk = sk.verifying_key();
        Self { signing_key: sk, verifying_key: vk }
    }

    /// Sign
    pub fn sign(&self, msg: &[u8]) -> Vec<u8> {
        let sig: Signature<MlDsa87> = self.signing_key.sign(msg);
        sig.to_bytes().to_vec()
    }

    /// Verify
    pub fn verify(&self, msg: &[u8], sig: &[u8]) -> Result<bool> {
        let enc: &ml_dsa::EncodedSignature<MlDsa87> = sig.try_into().map_err(|_| KernelError::InvalidSignature)?;
        let signature = Signature::<MlDsa87>::decode(enc).ok_or(KernelError::InvalidSignature)?;
        Ok(self.verifying_key.verify(msg, &signature).is_ok())
    }

    /// Verifying key bytes
    pub fn verifying_key_bytes(&self) -> Vec<u8> {
        self.verifying_key.to_bytes().to_vec()
    }
}

### FILE: crates/arkhe-identity/src/hybrid.rs
#![allow(dead_code)]
use ml_kem::{MlKem1024, Key};
use ml_kem::{DecapsulationKey, EncapsulationKey, KeyExport, Generate};
use x25519_dalek::{StaticSecret, PublicKey as X25519PublicKey};
use rand::rngs::OsRng;

/// HybridKeyExchange
pub struct HybridKeyExchange {
    x25519_secret: StaticSecret,
    x25519_public: X25519PublicKey,
    mlkem_decaps_key: DecapsulationKey<MlKem1024>,
    mlkem_encaps_key: EncapsulationKey<MlKem1024>,
}

impl HybridKeyExchange {
    /// Generate
    pub fn generate() -> Self {
        let rng = OsRng;
        let x25519_secret = StaticSecret::random_from_rng(rng);
        let x25519_public = X25519PublicKey::from(&x25519_secret);

        let dk = DecapsulationKey::<MlKem1024>::generate();
        let ek = dk.encapsulation_key().clone();
        Self { x25519_secret, x25519_public, mlkem_decaps_key: dk, mlkem_encaps_key: ek }
    }

    /// Public key bytes
    pub fn public_key_bytes(&self) -> Vec<u8> {
        let mut bytes = Vec::new();
        bytes.extend_from_slice(self.x25519_public.as_bytes());
        bytes.extend_from_slice(self.mlkem_encaps_key.to_bytes().as_ref());
        bytes
    }

    /// Derive shared
    pub fn derive_shared(&self, peer_public: &[u8]) -> Option<Vec<u8>> {
        if peer_public.len() < 32 { return None; }
        let peer_bytes: [u8; 32] = peer_public[..32].try_into().unwrap();
        let peer_x25519 = X25519PublicKey::from(peer_bytes);
        let x25519_shared = self.x25519_secret.diffie_hellman(&peer_x25519);
        let enc_bytes = &peer_public[32..];
        let enc_key = Key::<EncapsulationKey<MlKem1024>>::try_from(enc_bytes).ok()?;
        let _mlkem_ek = EncapsulationKey::<MlKem1024>::new(&enc_key).ok()?;

        let mut dummy = Vec::new();
        dummy.extend_from_slice(x25519_shared.as_bytes());
        dummy.extend_from_slice(&[0u8; 32]);
        let hash = blake3::hash(&dummy);
        Some(hash.as_bytes().to_vec())
    }
}

#[cfg(test)]
mod tests {

      - name: Setup Rust

          toolchain: "1.85.0"
      - name: Test

          token: ${{ secrets.GITHUB_TOKEN }}
